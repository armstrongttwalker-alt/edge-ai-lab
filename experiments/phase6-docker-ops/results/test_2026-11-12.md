## 启动命令

```bash
# Qdrant
sudo docker run -d --name edge-ai-qdrant --restart unless-stopped \
    -p 6333:6333 \
    -v /opt/edge-ai-lab/services/qdrant/storage:/qdrant/storage \
    docker.m.daocloud.io/qdrant/qdrant:latest

# Prometheus  
sudo docker run -d --name edge-ai-prometheus --restart unless-stopped \
    -p 9090:9090 \
    -v /opt/edge-ai-lab/services/monitoring/prometheus.yml:/etc/prometheus/prometheus.yml:ro \
    docker.m.daocloud.io/prom/prometheus:latest

# Grafana
sudo docker run -d --name edge-ai-grafana --restart unless-stopped \
    -p 3100:3000 -u root \
    -v /opt/edge-ai-lab/services/monitoring/grafana:/var/lib/grafana \
    -e GF_SECURITY_ADMIN_PASSWORD=edgeai \
    docker.m.daocloud.io/grafana/grafana:latest

# Open WebUI（待镜像下载完成）
sudo docker run -d --name edge-ai-webui --restart unless-stopped \
    -p 3000:8080 \
    -v /opt/edge-ai-lab/services/open-webui/data:/app/backend/data \
    -e OPENAI_API_BASE_URL=http://host-gateway:8000/v1 \
    --add-host=host-gateway:host-gateway \
    ghcr.m.daocloud.io/open-webui/open-webui:main
```
